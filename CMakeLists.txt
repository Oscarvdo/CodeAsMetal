cmake_minimum_required(VERSION 3.25)
# The supplied .slnx/.vcxproj delegates to this build graph; no Qt VS extension
# or manual per-machine linker lists are needed in the user's template.
project(CodeAsMetal VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CAM_BUILD_DESKTOP "Build Windows Qt/OCCT desktop" ON)
option(CAM_USE_GTEST "Build GoogleTest suites" ON)
include(CTest)
add_library(cam_core src/core/Domain.cpp)
target_include_directories(cam_core PUBLIC include)
if(MSVC)
  target_compile_options(cam_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(cam_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(cam_core_checks tests/Standalone.cpp)
  target_link_libraries(cam_core_checks PRIVATE cam_core)
  add_test(NAME core-behavior COMMAND cam_core_checks)
  if(CAM_USE_GTEST)
    find_package(GTest CONFIG REQUIRED)
    add_executable(cam_core_tests tests/CoreTests.cpp)
    target_link_libraries(cam_core_tests PRIVATE cam_core GTest::gtest_main)
    include(GoogleTest)
    gtest_discover_tests(cam_core_tests)
  endif()
endif()
if(CAM_BUILD_DESKTOP)
  if(NOT WIN32)
    message(FATAL_ERROR "Desktop viewer targets Windows. For portable core use -DCAM_BUILD_DESKTOP=OFF -DCAM_USE_GTEST=OFF")
  endif()
  enable_language(RC)
  find_package(Qt6 6.5 CONFIG REQUIRED COMPONENTS Widgets Concurrent Sql PrintSupport)
  find_package(OpenCASCADE CONFIG REQUIRED)
  if(DEFINED OpenCASCADE_MAJOR_VERSION AND OpenCASCADE_MAJOR_VERSION GREATER_EQUAL 8)
    message(FATAL_ERROR "This release targets OCCT 7.x APIs. Use the dependency instructions; OCCT 8 migration is not validated.")
  endif()
  add_library(cam_cad src/cad/Cad.cpp)
  target_include_directories(cam_cad SYSTEM PUBLIC ${OpenCASCADE_INCLUDE_DIR} ${OpenCASCADE_INCLUDE_DIRS})
  target_link_directories(cam_cad PUBLIC ${OpenCASCADE_LIBRARY_DIR})
  # OCCT 7.8 renamed exchange toolkits. Support both 7.6/7.7 and 7.8/7.9.
  if(TARGET TKDESTEP)
    set(CAM_EXCHANGE TKDESTEP TKDESTL)
  else()
    set(CAM_EXCHANGE TKSTEP TKSTEPBase TKSTEPAttr TKXSBase TKSTL)
  endif()
  target_link_libraries(cam_cad PUBLIC cam_core Qt6::Core TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep
    TKGeomAlgo TKTopAlgo TKPrim TKBO TKBool TKShHealing TKMesh ${CAM_EXCHANGE})
  add_library(cam_persistence src/persistence/Document.cpp src/persistence/SqlRepository.cpp)
  target_link_libraries(cam_persistence PUBLIC cam_core Qt6::Core Qt6::Sql)
  qt_add_executable(CodeAsMetal WIN32
    CodeAsMetal/CodeAsMetal.cpp CodeAsMetal/CodeAsMetal.rc
    src/desktop/MainWindow.cpp src/desktop/Viewer.cpp src/desktop/Logging.cpp)
  target_include_directories(CodeAsMetal PRIVATE CodeAsMetal)
  target_link_libraries(CodeAsMetal PRIVATE cam_cad cam_persistence Qt6::Widgets Qt6::Concurrent Qt6::PrintSupport
    TKService TKV3d TKOpenGl)
  target_compile_definitions(CodeAsMetal PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    foreach(t cam_cad cam_persistence CodeAsMetal)
      target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8)
      target_compile_definitions(${t} PRIVATE NOMINMAX)
    endforeach()
  endif()
  if(BUILD_TESTING AND CAM_USE_GTEST)
    add_executable(cam_adapter_tests tests/AdapterTests.cpp)
    target_link_libraries(cam_adapter_tests PRIVATE cam_cad cam_persistence GTest::gtest Qt6::Core)
    add_test(NAME cad-and-document COMMAND cam_adapter_tests)
  endif()
  install(TARGETS CodeAsMetal RUNTIME DESTINATION .)
  install(DIRECTORY docs data DESTINATION .)
  install(FILES README_ES.md THIRD_PARTY.md DESTINATION .)
endif()
